# IN[0]: Linea

from Autodesk.DesignScript.Geometry import Plane, Vector

linea = IN[0]

# Dirección del eje Z global
z_dir = Vector.ZAxis()

# Vector director de la línea
dir_linea = linea.Direction

# Para un plano paralelo a Z, buscamos que su normal sea perpendicular a Z,
# así que usamos el producto cruz entre la dirección de la línea y Z.
normal_plano = dir_linea.Cross(z_dir)

# Si la línea es paralela a Z, el cruce será un vector nulo.
if normal_plano.Length == 0:
    # En este caso, creamos un plano XY a la altura inicial de la línea
    punto = linea.StartPoint
    plano = Plane.ByOriginNormal(punto, z_dir)
else:
    # El plano se define por un punto de la línea y la normal calculada
    punto = linea.StartPoint
    plano = Plane.ByOriginNormal(punto, normal_plano)

OUT = plano
