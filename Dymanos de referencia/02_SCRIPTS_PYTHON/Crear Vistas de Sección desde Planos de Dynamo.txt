# --- 1. IMPORTACIONES Y CONFIGURACIÓN ---
import clr
clr.AddReference('RevitAPI')
from Autodesk.Revit.DB import *

clr.AddReference('RevitServices')
from RevitServices.Persistence import DocumentManager
from RevitServices.Transactions import TransactionManager

clr.AddReference('ProtoGeometry')
from Autodesk.DesignScript.Geometry import *

doc = DocumentManager.Instance.CurrentDBDocument

# --- 2. FUNCIONES AUXILIARES ---
def to_revit_xyz_internal(dynamo_point):
    """Convierte un punto de Dynamo a un punto XYZ de Revit en unidades internas (pies)."""
    x = UnitUtils.ConvertToInternalUnits(dynamo_point.X, UnitTypeId.Meters)
    y = UnitUtils.ConvertToInternalUnits(dynamo_point.Y, UnitTypeId.Meters)
    z = UnitUtils.ConvertToInternalUnits(dynamo_point.Z, UnitTypeId.Meters)
    return XYZ(x, y, z)
    
def to_revit_vector(dynamo_vector):
    """Convierte un vector de Dynamo a un vector XYZ de Revit."""
    return XYZ(dynamo_vector.X, dynamo_vector.Y, dynamo_vector.Z)

def get_unique_view_name(base_name):
    """Busca un nombre de vista único, añadiendo V2, V3, etc."""
    all_views = FilteredElementCollector(doc).OfClass(View).ToElements()
    existing_names = {v.Name for v in all_views}
    if base_name not in existing_names: return base_name
    version = 2
    while True:
        new_name = "{} V{}".format(base_name, version)
        if new_name not in existing_names: return new_name
        version += 1

def get_view_family_type_id():
    """Encuentra el ID del tipo de familia de vista de sección."""
    collector = FilteredElementCollector(doc).OfClass(ViewFamilyType)
    for vft in collector:
        if vft.ViewFamily == ViewFamily.Section:
            return vft.Id
    return None

# --- 3. OBTENER LISTAS DE ENTRADAS ---
dynamo_planes = IN[0]
view_widths_meters = IN[1]
view_heights_meters = IN[2]
view_depths_meters = IN[3]
base_view_names = IN[4]

# Lista para guardar las vistas creadas
vistas_creadas = []

# Obtener el ID del tipo de vista una sola vez
vft_id = get_view_family_type_id()

# --- 4. BUCLE PRINCIPAL PARA PROCESAR CADA PLANO ---
# Iniciar una única transacción para todo el proceso
TransactionManager.Instance.EnsureInTransaction(doc)

# Verificar que el tipo de vista exista antes de empezar el bucle
if vft_id:
    # El bucle recorre las listas usando un índice
    for i in range(len(dynamo_planes)):
        try:
            # Obtener los datos para la iteración actual
            dynamo_plane = dynamo_planes[i]
            view_width_meters = view_widths_meters[i]
            view_height_meters = view_heights_meters[i]
            view_depth_meters = view_depths_meters[i]
            base_view_name = base_view_names[i]

            # Convertir unidades para la iteración actual
            internal_width = UnitUtils.ConvertToInternalUnits(view_width_meters, UnitTypeId.Meters)
            internal_height = UnitUtils.ConvertToInternalUnits(view_height_meters, UnitTypeId.Meters)
            internal_depth = UnitUtils.ConvertToInternalUnits(view_depth_meters, UnitTypeId.Meters)

            # Geometría para la caja de la vista
            origin = to_revit_xyz_internal(dynamo_plane.Origin)
            view_direction = to_revit_vector(dynamo_plane.Normal)
            up_direction = to_revit_vector(dynamo_plane.YAxis)
            right_direction = to_revit_vector(dynamo_plane.XAxis)

            # Crear la transformación
            transform = Transform.Identity
            transform.Origin = origin
            transform.BasisX = right_direction
            transform.BasisY = up_direction
            transform.BasisZ = view_direction
            
            # Definir las dimensiones de la caja
            half_width = internal_width / 2.0
            half_height = internal_height / 2.0
            min_point = XYZ(-half_width, -half_height, 0)
            max_point = XYZ(half_width, half_height, internal_depth)

            # Crear la BoundingBox
            section_box = BoundingBoxXYZ()
            section_box.Transform = transform
            section_box.Min = min_point
            section_box.Max = max_point

            # Crear la vista de sección
            new_section_view = ViewSection.CreateSection(doc, vft_id, section_box)
            
            # Asignar el nombre único
            unique_name = get_unique_view_name(base_view_name)
            new_section_view.Name = unique_name
            
            # Añadir la vista recién creada a nuestra lista de resultados
            vistas_creadas.append(new_section_view)
        
        except Exception as e:
            # Si algo falla con una vista, lo añadimos a la salida para saberlo
            vistas_creadas.append("Error en el ítem {}: {}".format(i, str(e)))

# Finalizar la transacción
TransactionManager.Instance.TransactionTaskDone()

# Devolver la lista de vistas creadas (o errores)
OUT = vistas_creadas
