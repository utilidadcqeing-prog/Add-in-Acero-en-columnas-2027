from collections.abc import Sequence

def is_seq(x):
    return isinstance(x, Sequence) and not isinstance(x, (str, bytes))

def same_shape(a, b):
    """Verifica recursivamente que a y b tengan la misma estructura (listas/longitudes)."""
    if is_seq(a) and is_seq(b):
        if len(a) != len(b): return False
        for x, y in zip(a, b):
            if not same_shape(x, y): return False
        return True
    return not is_seq(a) and not is_seq(b)

def pick_default(default_value, path_idx):
    """
    Devuelve el valor por defecto para la rama actual.
    - Si default_value es escalar -> se usa ese mismo en todos lados.
    - Si default_value es anidado y su forma coincide con IN[0]/IN[1] -> se indexa con path_idx.
    """
    if not is_seq(default_value):
        return default_value
    # Navega por path_idx (lista de índices) si el usuario pasó un IN[2] anidado
    d = default_value
    for i in path_idx:
        d = d[i]
    return d

def chain_if(conditions, results, default_value, path_idx=None):
    """
    Aplica la cadena IF de izquierda a derecha:
      - Retorna el primer 'results[i]' cuyo 'conditions[i]' sea True.
      - Si ninguno es True, retorna 'default_value' (escalar o rama correspondiente).
    Preserva la estructura exterior cuando hay anidación.
    """
    if path_idx is None: path_idx = []

    # Casos escalares
    if not is_seq(conditions) and not is_seq(results):
        return results if bool(conditions) else pick_default(default_value, path_idx[:-1] if path_idx else [])

    # Validaciones de forma
    if is_seq(conditions) != is_seq(results):
        raise Exception("IN[0] e IN[1] deben tener la misma estructura (ambos listas o ambos escalares en cada rama).")

    if len(conditions) != len(results):
        raise Exception("IN[0] e IN[1] deben tener la misma longitud en cada nivel.")

    # ¿Es lista plana de booleanos?
    plana = all(not is_seq(c) for c in conditions)

    if plana:
        # Cadena IF estricta: evalúa en orden
        for i, c in enumerate(conditions):
            if bool(c):
                return results[i]
        # Ninguno fue True -> usar default
        return pick_default(default_value, path_idx)
    else:
        # Anidada: resolver por sublista preservando estructura
        out = []
        for i, (c_sub, r_sub) in enumerate(zip(conditions, results)):
            out.append(chain_if(c_sub, r_sub, default_value, path_idx + [i]))
        return out
