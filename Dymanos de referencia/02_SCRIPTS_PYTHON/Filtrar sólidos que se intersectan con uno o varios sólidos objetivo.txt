import clr
clr.AddReference('ProtoGeometry')
from Autodesk.DesignScript.Geometry import *

solidos = IN[0]
objetivos = IN[1]

# Asegura que objetivos sea una lista
if not isinstance(objetivos, list):
    objetivos = [objetivos]

# Función para verificar si un sólido interseca con alguno de los objetivos
def interseca_con_alguno(solido, objetivos):
    for obj in objetivos:
        if solido.DoesIntersect(obj):
            return True
    return False

# Aplica la verificación a todos los sólidos
intersectados = [s for s in solidos if interseca_con_alguno(s, objetivos)]

OUT = intersectados
