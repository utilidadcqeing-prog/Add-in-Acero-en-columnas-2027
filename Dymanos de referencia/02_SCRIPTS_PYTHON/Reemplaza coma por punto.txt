# IN[0] = lista de strings (puede ser anidada) o un string individual

data = IN[0]

def replace_commas(x):
    """Recorre la lista (anidada o no) y reemplaza , por . en los strings"""
    if isinstance(x, list):
        return [replace_commas(i) for i in x]
    elif isinstance(x, str):
        return x.replace(",", ".")
    else:
        return x

OUT = replace_commas(data)
