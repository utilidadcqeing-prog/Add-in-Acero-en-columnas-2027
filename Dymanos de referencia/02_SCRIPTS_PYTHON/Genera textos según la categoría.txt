# IN[1] = texto para "Cimentación estructural"
# IN[2] = texto para "Pilares estructurales"
# IN[3] = texto para "Armazón estructural"
# IN[4] = texto para "Muros"
# IN[5] = texto para "Suelos"
# IN[6] = texto para "Armadura estructural"
# IN[7] = texto para "Escaleras"
# OUT  = lista con misma estructura de IN[0], reemplazando cada categoría por su texto

# --- util: normalizar nombre para comparar ---
def normalize(s):
    # pasa a str, minúsculas y quita acentos y dobles espacios
    try:
        s = str(s)
    except:
        s = ""
    s = s.strip().lower()
    # quitar acentos básicos
    replace = {
        u"á":"a", u"é":"e", u"í":"i", u"ó":"o", u"ú":"u",
        u"ä":"a", u"ë":"e", u"ï":"i", u"ö":"o", u"ü":"u",
        u"ñ":"n"
    }
    for k,v in replace.items():
        s = s.replace(k, v)
    # colapsar espacios
    s = " ".join(s.split())
    return s

# --- mapeo categoría -> texto ---
map_text = {
    normalize("Cimentación estructural"): IN[1],
    normalize("Pilares estructurales"):   IN[2],
    normalize("Armazón estructural"):     IN[3],
    normalize("Muros"):                   IN[4],
    normalize("Suelos"):                  IN[5],
    normalize("Armadura estructural"):    IN[6],
    normalize("Escaleras"):               IN[7],
}

def cat_to_name(obj):
    # acepta DB.Category (tiene .Name) o str
    # cualquier otro tipo devolverá cadena vacía
    try:
        name = getattr(obj, "Name", obj)
        return str(name)
    except:
        return ""

def pick_text(cat_obj):
    key = normalize(cat_to_name(cat_obj))
    return map_text.get(key, "")  # si no está en el mapeo, devolver ""

def recurse_same_shape(x):
    if isinstance(x, list):
        return [recurse_same_shape(i) for i in x]
    else:
        return pick_text(x)

OUT = recurse_same_shape(IN[0])
