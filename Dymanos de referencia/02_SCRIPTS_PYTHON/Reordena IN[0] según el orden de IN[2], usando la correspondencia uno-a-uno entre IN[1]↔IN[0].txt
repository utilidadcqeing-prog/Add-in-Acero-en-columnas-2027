# IN[0]: lista plana de elementos (a reordenar)
# IN[1]: lista plana de referencia (mismo tamaño que IN[0])
# IN[2]: misma lista que IN[1] pero reordenada (mismo tamaño)

from collections import defaultdict, deque

A = IN[0]  # elementos a reordenar
B = IN[1]  # clave de referencia (alineada con A)
T = IN[2]  # nuevo orden de la misma clave B

# Validaciones básicas
if not (isinstance(A, list) and isinstance(B, list) and isinstance(T, list)):
    OUT = "Error: todas las entradas deben ser listas planas."
else:
    if not (len(A) == len(B) == len(T)):
        OUT = "Error: las tres listas deben tener el mismo tamaño."
    else:
        # Intentamos mapear por valor (hash); si algún elemento no es hasheable, caemos a identidad (id)
        use_hash = True
        idx_map_hash = defaultdict(deque)
        try:
            for i, v in enumerate(B):
                idx_map_hash[v].append(i)
        except TypeError:
            # Hay elementos no hasheables en B: usamos identidad de objeto
            use_hash = False
            idx_map_id = defaultdict(deque)
            for i, v in enumerate(B):
                idx_map_id[id(v)].append(i)

        # Reordenar A siguiendo el orden de T
        resultado = []
        try:
            if use_hash:
                for v in T:
                    if not idx_map_hash[v]:
                        raise KeyError("Elemento no encontrado o multiplicidad inconsistente.")
                    i = idx_map_hash[v].popleft()
                    resultado.append(A[i])
            else:
                for v in T:
                    key = id(v)
                    if not idx_map_id[key]:
                        raise KeyError("Elemento (por identidad) no encontrado o multiplicidad inconsistente.")
                    i = idx_map_id[key].popleft()
                    resultado.append(A[i])
            OUT = resultado
        except KeyError as e:
            OUT = "Error: IN[2] no es una reordenación válida de IN[1] (faltan/sobran elementos)."
