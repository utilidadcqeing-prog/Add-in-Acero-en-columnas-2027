# IN[0] = datos (lista plana o anidada)
# IN[1] = máscara (misma estructura y tamaño que IN[0]) con True/False
# IN[2] = valor de reemplazo

datos = IN[0]
mask  = IN[1]
valor = IN[2]

from collections.abc import Sequence

def es_seq(x):
    return isinstance(x, (list, tuple))

def estructura_igual(a, b):
    """Verificación rápida (opcional) de forma/longitud en paralelo."""
    if es_seq(a) and es_seq(b):
        if len(a) != len(b):
            return False
        return all(estructura_igual(x, y) for x, y in zip(a, b))
    # Si alguno no es secuencia, ambos deben ser escalares
    return not es_seq(a) and not es_seq(b)

def reemplazar_por_mascara(a, m, v):
    """Devuelve 'a' con los elementos donde 'm' es True reemplazados por 'v'."""
    if es_seq(a) and es_seq(m):
        # Conservamos listas (Dynamo usa listas); si llega tupla, devolvemos lista
        return [reemplazar_por_mascara(x, y, v) for x, y in zip(a, m)]
    else:
        # Caso escalar
        return v if bool(m) else a

# Validación opcional para errores claros
if not estructura_igual(datos, mask):
    OUT = "ERROR: IN[0] e IN[1] no comparten la misma estructura/tamaño."
else:
    OUT = reemplazar_por_mascara(datos, mask, valor)
